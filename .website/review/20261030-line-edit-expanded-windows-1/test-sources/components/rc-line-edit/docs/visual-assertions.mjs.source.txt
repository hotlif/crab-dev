import { expect } from "../../../test/visual/component-fixture.mjs";
import { assertContentGeometry } from "../../../test/visual/content-geometry.mjs";

export function matrixInput(scene, appearance) {
    return scene.getByRole("textbox", { name: new RegExp(` · ${appearance}$`) });
}

export async function focusByKeyboard(page, stateControl, input) {
    await stateControl.focus();
    for (let index = 0; index < 8; index += 1) {
        await page.keyboard.press("Tab");
        if (await input.evaluate((node) => globalThis.document.activeElement === node)) break;
    }
    await expect(input).toBeFocused();
    expect(await input.evaluate((node) => node.matches(":focus-visible"))).toBe(true);
}

/** Native text editing can scroll horizontally. Prove both logical ends and
 * the whole value remain reachable instead of requiring the input to expand. */
export async function assertCompleteDraft(input, draft) {
    await expect(input).toHaveValue(draft);
    await input.press("ControlOrMeta+Home");
    await expect
        .poll(() => input.evaluate((node) => ({ start: node.selectionStart, end: node.selectionEnd })))
        .toEqual({ start: 0, end: 0 });
    const start = await input.evaluate((node) => ({ left: node.scrollLeft, width: node.scrollWidth, clientWidth: node.clientWidth }));
    await input.press("ControlOrMeta+End");
    await expect
        .poll(() => input.evaluate((node) => ({ start: node.selectionStart, end: node.selectionEnd })))
        .toEqual({ start: draft.length, end: draft.length });
    if (start.width > start.clientWidth + 1) {
        await expect.poll(() => input.evaluate((node, left) => Math.abs(node.scrollLeft - left), start.left)).toBeGreaterThan(0);
    }
    const end = await input.evaluate((node) => ({ left: node.scrollLeft, width: node.scrollWidth, clientWidth: node.clientWidth }));
    await input.press("ControlOrMeta+A");
    await expect
        .poll(() => input.evaluate((node) => ({ start: node.selectionStart, end: node.selectionEnd })))
        .toEqual({ start: 0, end: draft.length });
    await expect(input).toHaveValue(draft);
    return { start, end, selectedValueLength: draft.length };
}

export async function assertLineEditGeometry(input, frame) {
    const control = input.locator("..");
    const field = control.locator("..");
    const buttons = [];
    for (const button of await field.getByRole("button").all()) {
        buttons.push({
            button: await assertContentGeometry(button, control, { checkText: false }),
            icon: await assertContentGeometry(button.locator("svg"), button, { checkText: false }),
        });
        await expect(button).toHaveAttribute("aria-controls", await input.getAttribute("id"));
    }
    const supporting = [];
    for (const text of await field.locator("[data-field-support] > span").all()) {
        supporting.push(await assertContentGeometry(text, field));
    }
    return {
        field: await assertContentGeometry(field, frame, { checkText: false }),
        control: await assertContentGeometry(control, field, { checkText: false }),
        input: await assertContentGeometry(input, control, { checkText: false, allowHorizontalScroll: true }),
        label: await assertContentGeometry(field.locator("[data-field-label-text]"), frame),
        supporting,
        buttons,
    };
}
